{{- $body := .Content -}}
{{- $body = $body | replaceRE "(?s)<script[^>]*>.*?</script>" "" -}}
{{- $body = $body | replaceRE "(?s)<a[^>]*href=\"([^\"]+)\"[^>]*>(.*?)</a>" "[$2]($1)" -}}
{{- $body = $body | replaceRE "(?s)<h1[^>]*>(.*?)</h1>" "\n# $1\n" -}}
{{- $body = $body | replaceRE "(?s)<h2[^>]*>(.*?)</h2>" "\n## $1\n" -}}
{{- $body = $body | replaceRE "(?s)<h3[^>]*>(.*?)</h3>" "\n### $1\n" -}}
{{- $body = $body | replaceRE "<li[^>]*>" "\n- " -}}
{{- $body = $body | replaceRE "</?(p|div|section|article|ul|ol|blockquote)[^>]*>|<br[^>]*>" "\n" -}}
{{- $body = $body | replaceRE "<[^>]+>" "" | htmlUnescape | replaceRE "\n{3,}" "\n\n" | replaceRE "- \n+" "- " | strings.TrimSpace -}}
{{- return ($body | safeHTML) -}}
